\begin{frame}{From factoring problem to crypgoraphy}
  \begin{itemize}
    \item Our goal is to turn the (believed) NP problem of factoring into useful cryptography.
    \item Despite all of the math we've gone through over the past months, there is still more to learn.
    \item We need to understand properties of modular arithmetic when the modulus is a composite number.
  \end{itemize}
  
\end{frame}

\begin{frame}{Overview}
  \begin{enumerate}
    \item For RSA the secret decryption key, $d$, is computed from a combination of public information and $\tot(N)$, totient of the public key, $N$.
    \item The totient of $N$, $\tot(N)$, is computed from the prime factors of $N$.
    \item The first part of this presentation is aimed to help you understand what Euler's totient function is.
    \item The second part of this presentation is aimed at showing how this is used in the RSA algorithm.
  \end{enumerate}
  
\end{frame}

\begin{frame}{Relatively prime}
  \begin{definition}[Coprime]
    Two integers $a$ and $b$ are \textbf{coprime} if and only if the largest integer that divides both of them is 1.

    Other ways to express the same concept include “$a$ and $b$ are relatively prime” and “$\gcd(a, b) = 1$”.
  \end{definition}
  \note[item]{When we are in the world of integers, we can use the word “divide” to mean integer division with no remainder.}
  \note[item]{We are in (a part of) the world of integers.}
\end{frame}

\begin{frame}{Examples for 15}
  \begin{example}
    \begin{itemize}
      \item 3 and 15 are \emph{not} coprime because 3 divides both of them.
      \item 10 and 15 are \emph{not} coprime because 5 divides both of them.
      \item 8 and 15 \emph{are} coprime because 1 is the largest number that divides both of them.
    \end{itemize}
  \end{example}
\end{frame}

\begin{frame}{The group $\Zn{15}$}
  \begin{definition}[{$\Zn{15}$}]
    The group $\Zn{15}$ is the set of elements,
    \set{1,2,4,7,8,11,13,14}, 
    along with the operation multiplication modulo~15. 
  \end{definition}
  \note[item]{Note that the members of the group do \emph{not} include 0, 3, 5, 6, 9, 10, and 12.}
\end{frame}

\begin{frame}{$\Zn{15}$ and its group properties}
  $\Zn{15}$ is an abelian (commutative) group because for all elements of the group $a$, $b$ and $c$
  \begin{itemize}
    \item It is \textbf{closed}.
      $a \times b$ is also in the group.
    \item There is an \textbf{identity element}, which in this case is 1, such that 
      $a \times 1 \equiv a$.
    \item Every element has an \textbf{inverse}:
      For every $a$ there is a member of the group that we will call `$a^{-1}$' such that $a \times a^{-1} \equiv 1$.
    \item The operation is commutative, making the group \textbf{abelian}:
      $a \times b \equiv b \times a$
    \item The operation is \textbf{associative}:
      $(a \times b) \times c \equiv a \times (b \times c)$.
  \end{itemize}
  \note[item]{By now, many of you should be able to recite the properties of a group in your sleep. And perhaps I have put you to sleep. But we also have newcomers.}
\end{frame}

\section{Totients}

\begin{frame}{How many coprimes?}
  \begin{itemize}
    \item There are eight numbers less than 15 that are coprime with 15.
    \item This is typically written “$\tot(15) = 8$’, using the small Greek letter ‘$\phi$’ (phi) or the variant form ‘$\varphi$’.
    \item $\tot(n)$ is called the “totient of $n$”.
  \end{itemize}
\end{frame}

\begin{frame}{Totient}
  \begin{definition}[Totient]
    The totient of a number $n$, written $\tot(n)$,
    is the number of numbers which are coprime with $n$ and less than $n$.
    \note[item]{The term “totient” was introduced in 1879 with no explanation by \textcite{Sylvester1879:Ternary:II}.
    Seems to be some sort of total on analogy to “quotient.”}
  \end{definition}
\end{frame}

\begin{frame}{Totient of 15}
  \begin{itemize}
    \item $\tot(15) = 8$
    \item $15 = 3 \times 5$
    \item $\tot(15) = (3 - 1)(5 - 1)$
    \item That is not a coincidence, and we will return to it.
  \end{itemize} 
\end{frame}

\section{Fermat's Little Theorem}

\begin{frame}{Little vs Last}
  \begin{itemize}
    \item Fermat's \emph{Last} Theorem is famous because he never proved it.
    \note[item]{The key to proving Fermat's \emph{Last} Theorem in 1994 rested on an important theorem about elliptic curves over rational fields.}
    \item Fermat never proved the “little” theorem either.
    \item Fermat's Little Theorem was extended and proved by Euler.
    \item Unlike Fermat's Last Theorem, Fermat's \emph{Little} Theorem is enormously useful.
  \end{itemize}
  
\end{frame}

\begin{frame}{Fermat's Little Theorem}
  \begin{theorem}[Fermat's Little Theorem]\label{t:flt1}
    If $p$ is prime and $a$ is not a multiple of $p$ then
    \[
        a^{p-1} \equiv 1 \pmod p
    \] 
\end{theorem}
\end{frame}

\begin{frame}{Multiply by $a$}
  \begin{align*}
      a^{p-1}         &\equiv 1 \pmod p
        && \mcomm{Theorem~\ref{t:flt1}} \\
      a^{p-1} \cdot a  &\equiv 1 \cdot a \pmod p

        && \mcomm{Exponentiation rules!}
  \end{align*}

\end{frame}

\begin{frame}{Also Fermat's Little Theorem}
  \begin{theorem}[Also Fermat's Little Theorem]\label{t:flt}

        a^{p} \equiv a \pmod p
    \] 
\end{theorem}
\end{frame}

\begin{frame}{Historical note}
  \begin{itemize}
    \item Chinese mathematicians were long aware of this, perhaps as far back as 500 BCE\@.
    \note[item]{There was some destruction of books (and killing of scholars)  during the Qin dynasty, but modern historians do not think it was anywhere as bad as Han dynasty reports made it out to be.}
    \item Fermat rediscovered it and said to have a proof in 1640.
    \note[item]{Not publishing a proof was common.
    It was a time of weird rivelries, a penchent for keeping discoveries secret, the fact that Fermet was an amateur who wrote things in letters to his friends, and there really wasn't a system for publishing proofs at the time.}
    \item Leibniz proved it before 1683 in an unpublished manuscript.
    \note[item]{There is a great deal of historical scholarship on what Leibniz knew and when.}
    \item Euler published a proof in 1740.
    \item Euler proved an important generalization of FLT in the same year.
    \note[item]{It is Euler's generalation that we go to next.}
  \end{itemize}
  
\end{frame}

\subsection{Euler's Totient Theorem}

\begin{frame}{Euler's extension}
  \begin{theorem}[Euler's Totient Theorem]\label{t:et}
    If $a$ and $n$ are coprime
    \[
      a^{\tot(n)} \equiv 1 \pmod n
    \] 
  \end{theorem}

\end{frame}

\begin{frame}{Multiplying Euler's Theorem}
  \begin{align*}
      a^{\tot(n)}         &\equiv 1 \pmod n \\
      a^{\tot(n)} \cdot a &\equiv 1 \cdot a \pmod n 
        && \mcomm{Multiply both sides by $a$}\\
      a^{\tot(n) + 1}     &\equiv a \pmod n
        && \mcomm{Exponentiation tricks}
  \end{align*}
\end{frame}

\begin{frame}{Also Euler's Theorem}
  \begin{theorem}[Also Euler's Totient Theorem]\label{t:eta}
    If $a$ and $n$ are coprime
    \[
      a^{\tot(n) + 1} \equiv a \pmod n
    \] 
  \end{theorem}
  
\end{frame}

\begin{frame}{Computing $\tot(n)$}
  \begin{definition}[Totient formula for $n = pq$; $p\neq q$]
    When $n = pq$ where $p$ and $q$ are distinct primes,
    $\tot(n) = (p - 1)(q -1)$.
  \end{definition}
  \note[item]{Euler provided a much more general formula for computing totients, but this simple case is all we need.
  In all cases, one needs the prime factors of $n$ to compute $\tot(n)$.}
\end{frame}

\section{RSA}


\begin{frame}{RSA public key}
  \begin{itemize}
    \item A prime public exponent, $e$, which is not unique to the user and is typically 65537.
    \note[item]{65537 is 0b10000000000000001. That is a lot of 0s and only two 1s. This means that the square-and-multiply algorithm only has to do two multiplies (and 16 squarings).}
    \note[item]{The square-and-multipy algorithm for exponentiation is 
      analogous to the double-and-add algorithm we went over
      when covering ECHD.}
    \note[item]{Because $e$ is a public modulus,
      we do not need to worry about side channel attacks revealing it,
      and so using square-and-multiply is fine.}
    \item A public modulus, $N$ which is unique to the user and is the product of two primes.
    \note[item]{The standards allow for public moduli to be constructed as the products of more than two primes. There were reasonable reasons at the time, but I don't think that any system has actually made use of that flexibility.}
  \end{itemize}
\end{frame}

\begin{frame}{RSA private key}
  \begin{itemize}
    \item Everything that is in the public key
    \item The primes $p$ and $q$ such that $N = pq$
    \item A decryption exponent, $d$, computed from $p$, $q$ and $e$.
    \note[item]{There are other things that are precomputed that are helpful for performing computations, we need to focus on $d$ here.}
  \end{itemize}
\end{frame}

\begin{frame}{The decryption exponent}
  \begin{definition}[The decryption secret]\label{d:d}
    The decryption exponent, $d$, is computed from
    the public exponent $e$ and the factors, $p$ and $q$, of
    public modulus, $N$.
    \[
    d = e^{-1} \bmod {\tot(N)}
    \]
  \end{definition}
  \note[item]{There is an efficient algorithm for computing the modular multiplicative inverse. And there are even quicker ones when you know the
  prime factors of the modulus as we do.}
\end{frame}

\begin{frame}{What does $a \equiv 1 \pmod{M}$ really mean?}
  “$a \equiv 1 \pmod M$” means (among other things)
  \begin{enumerate}
    \item If we divide $a$ by $M$ we get a remainder of 1;
    \item $a$ is one more than some multiple of $M$;
    \item There exists some integer $k$ such that $a = kM + 1$.
  \end{enumerate} 
  That last notion, $a = kM + 1$, is going to useful in what follows.
  \note[item]{Early on, I made a choice to use ‘$=$’ in all the places it should be used and also where ‘$\equiv$’ should be used.
  As I wasn't defining modular operations formally and pedantically, I thought I could get away with it. I am now paying the price, as the distinction between the two would be really useful here, but it would mean that I would have to use them correctly and consistently throughout.}
\end{frame}

\begin{frame}{This will be useful later}
A useful fact follows from how we defined $d$:
  \begin{equation}\label{eq:ed}
    \begin{split}
      ed &\equiv 1 \pmod{\tot(N)} \\
      ed &= k\tot(N) + 1 \qquad\text{(For some integer $k$)}
    \end{split}
  \end{equation}
\end{frame}

\begin{frame}{Encryption and Decryption}
  \begin{align}
    c &= m^e\label{eq:enc} \bmod N\\
    m &= c^d\label{eq:dec} \bmod N
  \end{align}
\end{frame}

\begin{frame}{Why does that work?}
    \begin{align*}
      c^d &\equiv {\left(m^e\right)}^d
            &&\mcomm{From $c \equiv m^e$} \\
          &\equiv m^{ed} 
              &&\mcomm{From how exponents work} \\
          &\equiv m^{k\tot(N) + 1}
              &&\mcomm{From Eq.~\ref{eq:ed} “a useful fact”}
              \tageq\label{eq:pre-miracle} \\
          &\equiv m 
              &&\mcomm{A miracle occurs}
    \end{align*}
    \note[item]{The miracle involves Euler's Theorem.}
\end{frame}

  \begin{figure}
    \includegraphics[width=0.5\textwidth]{Then-a-Miracle-Occurs-Copyrighted-artwork-by-Sydney-Harris-Inc-All-materials-used-with_W640.jpg}
    \caption[“Then a miracle happens”]
      {“Then a miracle occurs” Comic by Sidney Harris. Used by permission of \href{http://www.sciencecartoonsplus.com/pages/bio.php}{ScienceCartons}}
  \end{figure}
  \note[item]{Time for me to go into Certify to get reimbursed for the licensing.}
\end{frame}

\begin{frame}{The miracle}
  \note[item]{This slide is for additional material
  For those who want to see why I can ignore the $k$}

  \begin{align*}
    c^d &\equiv m^{k\tot(N) + 1}
          && \mcomm{Our pre-miracle equation (\ref{eq:pre-miracle})}\\
        &\equiv m^{k\tot(N)} \cdot m^1
          && \mcomm{From how exponents work} \\
        &\equiv \left({m^{\tot(N)}}\right)^k \cdot m^1
          && \mcomm{Also from how exponents work} \\
        &\equiv 1^k \cdot  m^1
          && \mcomm{From Euler's Theorem} \\
        &\equiv 1\cdot m 
          && \mcomm{From how 1 works} \\
        &\equiv m
          && \mcomm{Also from how 1 works} 
  \end{align*}

  \note[item]{I hope that you all see why I did Diffie-Hellman first. It gave everyone practice with modular exponentiation in smaller bites.}
\end{frame}

\section{How strong is RSA?}

\subsection{Key recovery}

\begin{frame}{Key recovery is as hard as factoring}
  \begin{enumerate}
    \item There is a polynomial time algorithm for computing $d$ from the factors of $N$.
    \note[item]{It is simply how we computed $d$ in the first place.}

    \item\label{en:kr:f2d} Therefore key recovery is not harder than factoring.
    \note[item]{Another way to say $X$ is not harder than $Y$ is to say $Y$ is at least as hard as $X$. So factoring is at least as hard as key recovery.}
    
    \item There is a polynomial time algorithm for computing the factors of $N$ from $d$.
    \note[item]{Computing the factors of $N$ from $d$ and $e$ involves a messy algorithm. It would be tedious to describe and show why it works. But it exists and is \spoly[N].}

    \item\label{en:kr:d2f} Therefore factoring is not harder than key recovery.
    
    \item From \ref{en:kr:f2d}~and~\ref{en:kr:d2f} we know that RSA key recovery and factoring are equally hard.
  \end{enumerate}
  
\end{frame}

\begin{frame}{Does this mean that breaking RSA is as hard as factoring?}
  \begin{description}
    \item[Q:] Does this mean that breaking RSA is as hard as factoring?
    \item[A:] No! Key recovery is not the only way to break a cipher scheme. 
  \end{description} 
\end{frame}

\subsection{Security notions}

\begin{frame}{Notions again}
  \note[item]{We went through notions and \indcpa\ more than half a year ago,
    so this is a brief recap.}
  \begin{itemize}
    \item We don't know all the ways to attack an encryption scheme.
    
    \item We can only list the ways we know of to attack an encryption scheme.
    \note[item]{It is very important to demostrate that a scheme is secure against such attacks.}

    \item But we can define what it means for a scheme to be broken irrespective of how it is broken.

    \item An encryption scheme is broken if having the ciphertext reveals new information about the content of the plaintext.
    \note[item]{“New information” is defined in a way that makes traffic analysis somebody else's problem.}
    
    \item In practice these definitions are in terms of adversaries playing well-defined games against the scheme.
    \note[item]{The proof that the \indcpa\ game definition and the “ciphertext provides no new information” definitions are the same is left as an excercie to the reader.}
    \note[item]{Or you could take my word for it that there is such a proof.}
  \end{itemize}
\end{frame}

\begin{frame}{\indcpa\ game: The RSA edition}
  The adversary, \adv, plays a game against the challenger that goes like this
  \begin{enumerate}
    \item During setup, the challenger generates a random RSA key pair. The public key, \pk, is given to \adv.
    \item The challenger flips a fair coin to set a bit, $b$, to either 0 or 1. $b$ is not given to \adv.
    \item \adv\ creates two plaintexts of its choosing, $m_0$ and $m_1$.
    \item The challenger encrypts $m_0$ or $m_1$ depending on the random bit $b$ and returns the ciphertext $c_b$ to \adv.
    \item \adv\ studies what it has \set{\pk, m_0, m_1, c_b} to decide which of $m_0$ or $m_1$ was encrypted.
    \item If \adv's conclusion is correct \adv\ wins.
  \end{enumerate}
  \note[item]{This game captures the notion that the ciphertext does not give an attacker any new information about the content of the plaintext.}
\end{frame}

\begin{frame}{RSA Assumption}
  \begin{definition}[RSA Assumption]
    No polynomial time algorithm can win the \indcpa\ RSA game (meaningfully) better than half the time.
  \end{definition} 
\end{frame}

\begin{frame}{Proving in the right direction}
  \begin{description}
    \item[Tired] If you can break problem $P$, you can break my cipher scheme. Therefore my cipher scheme is as hard as $P$
    \item[Wired] If you can break my cipher scheme, you can break problem $P$. Therefore my scheme is at least as hard as $P$.
  \end{description}
\end{frame}

\begin{frame}{Is breaking RSA as hard as factoring?}
  \begin{description}
    \item[Q:] Is breaking RSA as hard as factoring?
    \item[A1:] For “textbook RSA” described so far, the answer is no.
    \item[A2:] For non-deterministic RSA the answer is that it is complicated.
    \note[item]{The answer is probably “no” for small public exponent $e$. For suffiently large $e$ the answer is “we hope so.”} 
  \end{description}
\end{frame}

\subsection{Non-deterministic encryption}

\begin{frame}{Encryption must be randomized}
  As described so far, \adv\ can win the \indcpa\ game 100\% of the time. \adv\ can simply encrypt $m_0$ and $m_1$ with the public key to generate $c_0$ and $c_1$, one of which will be equal to the challenge ciphertext $c_b$. 
\end{frame}

\begin{frame}{OAEP}
  \begin{itemize}
    \item Optimal Asymmetric Encryption Padding (OAEP)
    \item Does the job analogous to the padding used in block ciphers
    \item Does the job analogous to the nonce/IV used in block ciphers
    \item It does some hashing with the randomness to make sure that the bit sequence (other than first two bits) that goes through actual RSA has no usable structure.
  \end{itemize}
\end{frame}

\begin{frame}{Avoid PKCS1 -- Use OAEP}
  \begin{itemize}
    \item In addition to OAEP, there is another RSA padding scheme, PKCS1.
    \item PKCS1 does almost everything right, but OAEP is superior in every respect.
    \item Many cryptographic libraries offer both PKCS1 and OAEP modes.
    \note[item]{crypto libraries really should do more to discourage PKCS1. One way would be to make it unavailable when creating new messages.}
    \item Do not use PKCS1 unless you have to.
    \item If you must use PKCS1, be aware that you can sometimes get a “successful” decryption of mangled data.
  \end{itemize}
  
\end{frame}
\subsection{What we know about RSA}

\begin{frame}{Factoring assumption}
  \begin{definition}[Factoring assumption]
    There is no $\spoly[x]$ algorithm which can factor an appropriately chosen $x$. 
  \end{definition}
\end{frame}

\begin{frame}{What a proof would look like}
    \begin{enumerate}
      \item Imagine that a polynomial time algorithm, \adv, has access to an oracle that has broken RSA encryption.
      \note[item]{Anyone else thinking of Anselm of Cambridge's ontological argument? Well what we are doing here really is solid and works.}
      \item We don't have any knowledge of the internal workings of the oracle, but we can feed it ciphertexts and get out plaintexts.
      \item Can \adv\ use its freedom to get decryptions from the oracle to factor the public key?
      \item If yes, then the RSA problem is at least as hard as factoring.
  \end{enumerate}
  So far no such proof exists.
  \note[item]{It is unlikely that such a proof exists for the RSA assumption as it stands. There are special cases, such as for certain values of $e$, where a polynomial algorithm does exist. So the RSA assumption will need refining.}
\end{frame}

\begin{frame}{Euler's totient theorem again}
  In a mod $m$ world
  \begin{align*}
    aa^{-1}       &\equiv 1 &&\mcomm{Definition of $a^{-1}$} \\
    a^{\tot(m)}   &\equiv 1  && \mcomm{Euler's theorem} \\
    aa^{\tot(m)-1}  &\equiv 1 && \mcomm{How exponents work}\\
    aa^{\tot(m)-1}  &\equiv aa^{-1} && \mcomm{$a$ times its inverse is 1} \\
    a^{\tot(m) -1} &\equiv a^{-1}  && \mcomm{Multiply by $a^{-1}$ (divide by $a$)}
  \end{align*}
  \note[item]{The intuition (at least for me) is that if $\tot(m)$ number of $a$s multiplied together take you around to 1
  then $\tot(m) - 1$ number of  $a$s take you to the thing you could multiply by one more $a$ to get to 1.
  That thing that you multiply “one more $a$ to to get 1 is the inverse of $a$.}
  \note[item]{On “how exponents work”: $a^x = aa^{x-1}$. }
  
\end{frame}
